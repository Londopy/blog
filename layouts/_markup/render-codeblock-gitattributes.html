{{- /* Chroma has no .gitattributes lexer, so fences tagged `gitattributes` are tokenized here and emitted with Chroma's class names. The colors then come from the same stylesheet as every other code block, and PaperMod's copy button finds the block the same way. # comment c1 [attr]name k + nf (macro definition) pattern s attr, -attr, !attr na, with the prefix as o attr=value na o l */ -}} 

{{- range split (strings.TrimSuffix "\n" .Inner) "\n" -}}

  {{- if hasPrefix (strings.TrimSpace .) "#" -}}
    {{ . }}
  {{- else -}}
    {{- $first := true -}}
    {{- range findRE `\s+|\S+` . -}}
      {{- if eq (strings.TrimSpace .) "" -}}
        {{ . }}
      {{- else if $first -}}
        {{- $first = false -}}
        {{- if hasPrefix . "[attr]" -}}
          [attr]{{ strings.TrimPrefix "[attr]" . }}
        {{- else -}}
          {{ . }}
        {{- end -}}
      {{- else if or (hasPrefix . "-") (hasPrefix . "!") -}}
        {{ substr . 0 1 }}{{ substr . 1 }}
      {{- else if in . "=" -}}
        {{- $kv := split . "=" -}}
        {{ index $kv 0 }}={{ delimit (after 1 $kv) "=" }}
      {{- else -}}
        {{ . }}
      {{- end -}}
    {{- end -}}
  {{- end -}}
{{ "\n" }}
{{- end -}}
